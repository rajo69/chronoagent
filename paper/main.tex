% Guarded \input: only include a .tex fragment if it exists on disk. Keeps
% the paper compilable before the Phase 10 results are regenerated.
\newcommand{\inputiffound}[2]{%
  \IfFileExists{#1}{\input{#1}}{%
    \par\noindent\textit{#2 (\texttt{#1}, from \texttt{make reproduce}).}\par%
  }%
}

% Guarded \includegraphics: include the figure at the given explicit path
% if it exists, otherwise emit a line naming the artefact. Same rationale as
% \inputiffound -- Phase 10 figures live under ../results/ and may be
% absent on a clean checkout, so a bare \includegraphics would hard-fail
% the PDF build. #1 is the full path (with .png) relative to the paper/
% directory so \IfFileExists can resolve it; \graphicspath is not used
% here because \IfFileExists ignores it.
\newcommand{\figureiffound}[2]{%
  \IfFileExists{#1}{\includegraphics[width=\columnwidth]{#1}}{%
    \textit{#2 (\texttt{#1}, from \texttt{make reproduce}).}%
  }%
}
